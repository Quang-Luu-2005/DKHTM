\section{Chức năng đã hoàn thành \& hoạt động ổn định}
%============================

\begin{frame}{List chức năng đã hoàn thành (1/2)}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{Điều khiển cổng \& Servo}
      \begin{itemize}
        \item \textbf{Điều khiển cổng:} Chế độ bình thường, mở cưỡng bức, khóa cưỡng bức (Đã lập trình).
        \item \textbf{Servo:} Tự động mở cổng khi có quyền và tự đóng (Đã lập trình).
      \end{itemize}
      \vspace{0.4em}
      \textbf{Định danh RFID / NFC}
      \begin{itemize}
        \item \textbf{RFID đọc UID:} Gửi lên Server đối chiếu Database (Đã hoạt động).
        \item \textbf{RFID hợp lệ:} Mở cổng, LED xanh, buzzer báo, ghi log (Đã hoạt động).
        \item \textbf{RFID không hợp lệ:} Đóng cổng, LED đỏ, buzzer kêu 3 lần (Đã lập trình).
        \item \textbf{Liên kết thẻ:} Quét và gán thẻ cho nhân viên (Đã hoạt động).
      \end{itemize}

    \column{0.48\textwidth}
      \textbf{Hạ tầng Kết nối \& Web Sentinel}
      \begin{itemize}
        \item \textbf{ESP32--HiveMQ:} Wi-Fi, TLS MQTT, gửi/nhận lệnh (Đã hoạt động).
        \item \textbf{Web Sentinel:} Giao diện điều khiển & nhận trạng thái qua MQTT bridge (Đã hoạt động).
        \item \textbf{Nhật ký (Log):} Ghi nhận lượt RFID, khuôn mặt, từ chối & vi phạm (Đã lập trình).
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{List chức năng đã hoàn thành (2/2)}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{Quản lý \& Thông báo Email}
      \begin{itemize}
        \item \textbf{Quản lý nhân viên:} Thêm/sửa nhân viên, vai trò (role), email, UID RFID và ảnh khuôn mặt (Đã lập trình).
        \item \textbf{Email cơ bản:} Tự động gửi email xác nhận cho nhân viên khi lượt ra vào được chấp nhận (Đã hoạt động).
        \item \textbf{Email vi phạm:} Bộ lọc tự động gửi email cảnh báo chỉ tới \textit{Administrator} và \textit{Security Officer} (Đã lập trình).
      \end{itemize}

    \column{0.48\textwidth}
      \textbf{Xử lý vi phạm \& Camera}
      \begin{itemize}
        \item \textbf{Xử lý vi phạm:} Khóa cổng, LED đỏ, buzzer kêu liên tục & đẩy sự kiện lên Web (Đã lập trình).
        \item \textbf{Reset vi phạm:} Mở khóa từ Web Sentinel hoặc nút bấm cứng trên mạch (Đã lập trình).
        \item \textbf{ESP32-CAM:} Capture snapshot, MJPEG stream, bật/tắt đèn trợ sáng (Đã lập trình).
        \item \textbf{Nhận diện khuôn mặt:} Giữ mặt 2s, so khớp ảnh, gửi kết quả về cổng (Đã lập trình).
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Minh chứng chức năng hoạt động ổn định}
  \begin{itemize}
    \item \textbf{Hệ thống Web \& API:} Web Sentinel & Node API chạy tại port 3000 và 3001, kết nối thành công với HiveMQ Cloud.
    \item \textbf{Giao tiếp Real-time:} Gửi/nhận dữ liệu ổn định qua 2 topic: \texttt{/board/upload/data} và \texttt{/board/get/data}.
    \item \textbf{Xác thực \& Lưu trữ:} Khởi động ESP32 tự kết nối Wi-Fi/MQTT, RFID hợp lệ mở cổng & ghi log, Database tự lưu và phục hồi dữ liệu nhân viên sau khi restart server.
    \item \textbf{Firmware \& Cấu hình:} Đã build và upload firmware thành công cho cả ESP32 Cổng và ESP32-CAM.
  \end{itemize}
\end{frame}

      \textbf{Chức năng chưa ổn định}
      \begin{itemize}
        \item \textbf{Stream ESP32-CAM:} Thỉnh thoảng bị đen hình, rò rỉ kết nối MJPEG.
        \item \textbf{Nhận diện khuôn mặt:} Chưa kiểm thử đa dạng góc mặt và điều kiện ánh sáng.
        \item \textbf{Phát hiện vi phạm:} Mới nạp firmware, chưa kiểm thử vật lý end-to-end với HC-SR04.
        \item \textbf{Phân quyền Email:} Database chưa cập nhật đủ email của Admin/Security Officer.
      \end{itemize}

    \column{0.48\textwidth}
      \textbf{Rủi ro phần cứng \& Vận hành}
      \begin{itemize}
        \item \textbf{Buzzer \& Nguồn:} Âm lượng buzzer chưa đồng đều do phụ thuộc nguồn điện.
        \item \textbf{Tự đóng cổng:} Dùng thời gian cố định (\textit{fixed delay}), chưa có cảm biến nhận biết người đã đi qua hoàn toàn.
        \item \textbf{Mạng \& Kết nối:} Chưa hỗ trợ stream khi MacBook và ESP32-CAM khác subnet; chưa kiểm thử mất kết nối Wi-Fi/MQTT dài hạn.
      \end{itemize}
  \end{columns}
\end{frame}

\section{Đề xuất hoàn thiện trong Sprint 6}
%============================

\begin{frame}{Kế hoạch hành động Sprint 6 (1/2)}
  \begin{enumerate}
    \item \textbf{Ổn định luồng Camera:}
      \begin{itemize}
        \item Tự động kết nối lại khi ESP32-CAM reset; triệt tiêu kết nối MJPEG trùng lặp.
        \item Đảm bảo stream chạy liên tục 30--60 phút không đen hình; hiển thị rõ trạng thái mất kết nối.
      \end{itemize}
    \vspace{0.3em}
    \item \textbf{Hoàn thiện Nhận diện khuôn mặt:}
      \begin{itemize}
        \item Kiểm thử nhiều nhân viên, ánh sáng khác nhau; thiết lập ngưỡng tin cậy (\textit{confidence threshold}) tối ưu.
        \item Mặt hợp lệ $\rightarrow$ Mở cổng; Mặt không hợp lệ $\rightarrow$ Báo lỗi 3 tiếng; Chống gửi lặp kết quả.
      \end{itemize}
    \vspace{0.3em}
    \item \textbf{Hoàn thiện Phát hiện vi phạm (HC-SR04):}
      \begin{itemize}
        \item Hiệu chỉnh khoảng cách và số lần xác nhận; tránh báo nhầm người vừa được cấp quyền ra vào.
        \item Vi phạm $\rightarrow$ Khóa cổng, bật buzzer liên tục, ghi log và gửi email tức thì.
      \end{itemize}
  \end{enumerate}
\end{frame}

\begin{frame}{Kế hoạch hành động Sprint 6 (2/2)}
  \begin{enumerate}
    \setcounter{enumi}{3}
    \item \textbf{Hoàn thiện Phân quyền Thông báo Email:}
      \begin{itemize}
        \item Cập nhật đầy đủ email Role Admin \& Security Officer.
        \item Phân luồng: Cảnh báo vi phạm chỉ gửi cho Admin/An ninh; Nhân viên chỉ nhận log ra vào cá nhân.
      \end{itemize}
    \vspace{0.3em}
    \item \textbf{Cải thiện State Machine của Cổng:}
      \begin{itemize}
        \item Loại bỏ hàm \texttt{delay()} gây nghẽn chương trình; chỉ đóng cổng khi người đã qua hoặc hết timeout an toàn.
        \item Tự khôi phục trạng thái chuẩn xác sau sự cố mất điện / mất mạng Wi-Fi / MQTT.
      \end{itemize}
    \vspace{0.3em}
    \item \textbf{Nhật ký \& Kiểm thử End-to-End:}
      \begin{itemize}
        \item Lưu log phía Server tránh mất data khi reload; lập bảng kết quả Pass/Fail và quay video demo.
      \end{itemize}
  \end{enumerate}
\end{frame}
